import { NextRequest } from "next/server";
import { prisma } from "@/lib/db";
import { confirmarPorSessao, descartarPorSessao, JANELA_RESERVA_MS } from "@/lib/coupon-redemption";

const POST_previa = require("@/app/api/patient/coupons/preview/route").POST;
const POST_consulta = require("@/app/api/patient/appointments/[id]/checkout/route").POST;
const POST_pacote = require("@/app/api/patient/packages/checkout/route").POST;
const POST_plano = require("@/app/api/patient/treatment-plans/checkout/route").POST;
const POST_assinatura = require("@/app/api/patient/membership/subscribe/route").POST;

const req = (url: string, body: any) =>
  new NextRequest(url, {
    method: "POST",
    body: JSON.stringify(body),
    headers: { "content-type": "application/json", "x-platform": "mobile" },
  });

const consulta = async (id: string, body: any) => {
  const r = await POST_consulta(req(`http://localhost:4100/api/patient/appointments/${id}/checkout`, body), {
    params: { id },
  });
  return { status: r.status, body: await r.json() };
};
const plano = async (body: any) => {
  const r = await POST_plano(req("http://localhost:4100/api/patient/treatment-plans/checkout", body));
  return { status: r.status, body: await r.json() };
};
const pacote = async (body: any) => {
  const r = await POST_pacote(req("http://localhost:4100/api/patient/packages/checkout", body));
  return { status: r.status, body: await r.json() };
};
const assinatura = async (body: any) => {
  const r = await POST_assinatura(req("http://localhost:4100/api/patient/membership/subscribe", body));
  return { status: r.status, body: await r.json() };
};
const previa = async (body: any) => {
  const r = await POST_previa(req("http://localhost:4100/api/patient/coupons/preview", body));
  return { status: r.status, body: await r.json() };
};

const unitAmount = (s: any) => s?.line_items?.[0]?.price_data?.unit_amount ?? null;

/** So o que pertence a clinica de teste: o Postgres local e compartilhado entre worktrees. */
const resgates = (code?: string) =>
  prisma.couponRedemption.findMany({
    where: { coupon: { clinicId: dados.clinicId, ...(code ? { code } : {}) } },
    select: {
      id: true,
      patientId: true,
      scope: true,
      originalAmount: true,
      discountAmount: true,
      finalAmount: true,
      stripeSessionId: true,
      confirmedAt: true,
      createdAt: true,
      coupon: { select: { code: true } },
    },
    orderBy: { createdAt: "asc" },
  });
const resumo = (rs: any[]) =>
  rs.map((r) => ({
    code: r.coupon.code,
    quem: r.patientId === A ? "A" : r.patientId === B ? "B" : r.patientId,
    escopo: r.scope,
    orig: r.originalAmount,
    desc: r.discountAmount,
    final: r.finalAmount,
    sessao: r.stripeSessionId,
    confirmado: r.confirmedAt ? "sim" : "nao",
  }));
const limpar = () => prisma.couponRedemption.deleteMany({ where: { coupon: { clinicId: dados.clinicId } } });

const statusConsulta = async (id: string) => {
  const a = await prisma.appointment.findUnique({ where: { id }, select: { status: true, price: true } });
  return a;
};
const voltarParaPendente = (id: string) =>
  prisma.appointment.update({ where: { id }, data: { status: "PENDING" } });

const log = (titulo: string, o: any) => console.log(`\n### ${titulo}\n${JSON.stringify(o, null, 1)}`);

beforeEach(async () => {
  await limpar();
  sessoes.length = 0;
  cuponsStripe.length = 0;
  cupomExiste = false;
  atual = A;
});

afterAll(async () => {
  await limpar();
  // Todas as consultas de teste voltam a PENDING para o QA poder ser repetido.
  for (const id of dados.consultasA) await voltarParaPendente(id).catch(() => {});
  await voltarParaPendente(dados.consultaB).catch(() => {});
  await prisma.$disconnect();
});

describe("D1 · A-1 — o segundo pedido de pagamento do mesmo paciente", () => {
  it("D1a · maxPerPatient:1 — 2o toque funciona e REAPROVEITA a reserva (1 resgate, sem already_used)", async () => {
    const id = dados.consultasA[0];
    const t1 = await consulta(id, { couponCode: "QA084-RT-PER1" });
    const t2 = await consulta(id, { couponCode: "QA084-RT-PER1" });
    const rs = await resgates("QA084-RT-PER1");
    log("D1a  dois toques, maxPerPatient:1, maxRedemptions:null", {
      toque1: { status: t1.status, body: t1.body },
      toque2: { status: t2.status, body: t2.body },
      unit_amounts: sessoes.map(unitAmount),
      sessoes: sessoes.map((_, i) => `cs_rt_${i + 1}`),
      linhas_de_resgate: resumo(rs),
      mesma_linha_reaproveitada: rs.length === 1,
    });
    expect(t1.status).toBe(200);
    expect(t2.status).toBe(200);
    expect(JSON.stringify(t2.body)).not.toContain("already_used");
    expect(rs).toHaveLength(1);
    expect(rs[0].stripeSessionId).toBe("cs_rt_2");
    expect(sessoes.map(unitAmount)).toEqual([8000, 8000]);
  });

  it("D1a2 · maxPerPatient:1 E maxRedemptions:1 — o 2o toque do mesmo paciente tambem passa", async () => {
    const id = dados.consultasA[1];
    const t1 = await consulta(id, { couponCode: "QA084-ONE" });
    const t2 = await consulta(id, { couponCode: "QA084-ONE" });
    const rs = await resgates("QA084-ONE");
    log("D1a2 dois toques no cupom de uso unico (maxRed 1, maxPer 1)", {
      toque1: { status: t1.status, body: t1.body },
      toque2: { status: t2.status, body: t2.body },
      linhas_de_resgate: resumo(rs),
    });
    expect(t1.status).toBe(200);
    expect(t2.status).toBe(200);
    expect(rs).toHaveLength(1);
  });

  it("D1b · maxRedemptions:1 — o SEGUNDO paciente e bloqueado pela reserva em andamento do primeiro", async () => {
    atual = A;
    const rA = await consulta(dados.consultasA[2], { couponCode: "QA084-ONE" });
    const reservaDeA = await resgates("QA084-ONE");

    atual = B;
    const pB = await previa({ code: "QA084-ONE", scope: "CONSULTATION" });
    const rB = await consulta(dados.consultaB, { couponCode: "QA084-ONE" });
    const depois = await resgates("QA084-ONE");
    log("D1b  A reserva (nao confirmado) e B tenta", {
      A_checkout: { status: rA.status, body: rA.body },
      reserva_de_A: resumo(reservaDeA),
      B_previa: { status: pB.status, body: pB.body },
      B_checkout: { status: rB.status, body: rB.body },
      resgates_ao_fim: resumo(depois),
    });
    expect(rA.status).toBe(200);
    expect(rB.status).toBe(409);
    expect(rB.body.reason).toBe("limit_reached");
    expect(depois).toHaveLength(1);
    expect(depois[0].patientId).toBe(A);
  });

  it("D1c · o limite DA PESSOA conta o confirmado: com o resgate pago, o proprio A e recusado", async () => {
    atual = A;
    const t1 = await consulta(dados.consultasA[3], { couponCode: "QA084-RT-PER1" });
    const confirmadas = await confirmarPorSessao("cs_rt_1");
    const t2 = await consulta(dados.consultasA[4], { couponCode: "QA084-RT-PER1" });
    const rs = await resgates("QA084-RT-PER1");
    log("D1c  A paga e tenta de novo", {
      toque1: { status: t1.status, sessao: "cs_rt_1" },
      confirmarPorSessao: confirmadas,
      toque2_depois_de_pago: { status: t2.status, body: t2.body },
      resgates: resumo(rs),
    });
    expect(confirmadas).toBe(1);
    expect(t2.status).toBe(409);
    expect(t2.body.reason).toBe("already_used");
    expect(rs).toHaveLength(1);
  });

  it("D1d · N-1 (a): o mesmo paciente NAO consegue o mesmo cupom de uso unico em dois escopos", async () => {
    // QA084-RT-DOIS: maxRedemptions 1, maxPerPatient 1, vale em CONSULTATION e MEMBERSHIP.
    atual = A;
    const c1 = await consulta(dados.consultasA[5], { couponCode: "QA084-RT-DOIS" });
    const a1 = await assinatura({ planId: dados.planoAssinatura, couponCode: "QA084-RT-DOIS" });
    const rs = await resgates("QA084-RT-DOIS");
    log("D1d  consulta e assinatura com o mesmo cupom de uso unico", {
      consulta: { status: c1.status, body: c1.body },
      assinatura: { status: a1.status, body: a1.body },
      resgates: resumo(rs),
      sessoes_criadas: sessoes.length,
    });
    expect(c1.status).toBe(200);
    expect(a1.status).toBe(409);
    expect(rs).toHaveLength(1);
  });

  it("D1e · N-1 (b): a janela da reserva e de 24h — igual a vida da sessao do Stripe", async () => {
    const horas = JANELA_RESERVA_MS / 3600000;
    atual = A;
    const t1 = await consulta(dados.consultasA[6], { couponCode: "QA084-ONE" });
    const r1 = await resgates("QA084-ONE");

    // Reserva envelhecida 23h: ainda dentro da janela -> outro paciente barrado.
    await prisma.couponRedemption.update({
      where: { id: r1[0].id },
      data: { createdAt: new Date(Date.now() - 23 * 3600000) },
    });
    atual = B;
    const b23 = await consulta(dados.consultaB, { couponCode: "QA084-ONE" });

    // Reserva envelhecida 25h: fora da janela -> a vaga volta (e a sessao do
    // Stripe ja morreu, porque uma sessao de Checkout vive 24h).
    await prisma.couponRedemption.update({
      where: { id: r1[0].id },
      data: { createdAt: new Date(Date.now() - 25 * 3600000) },
    });
    const b25 = await consulta(dados.consultaB, { couponCode: "QA084-ONE" });
    log("D1e  a janela da reserva", {
      JANELA_RESERVA_MS: `${JANELA_RESERVA_MS} ms = ${horas}h`,
      A_reservou: t1.status,
      B_com_reserva_de_23h: { status: b23.status, body: b23.body },
      B_com_reserva_de_25h: { status: b25.status, body: b25.body },
      resgates: resumo(await resgates("QA084-ONE")),
    });
    expect(horas).toBe(24);
    expect(b23.status).toBe(409);
    expect(b23.body.reason).toBe("limit_reached");
  });
});
