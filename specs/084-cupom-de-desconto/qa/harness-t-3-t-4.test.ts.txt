const dados = require("./dados.json");

const A = dados.patientA.id;
const B = dados.patientB.id;

// ── A Stripe, gravada ───────────────────────────────────────────────────────
const sessoes: any[] = [];
const cuponsStripe: any[] = [];
let falharSessao = false;
let cupomExiste = false;

(globalThis as any).__qa = {
  checkout: {
    sessions: {
      create: async (params: any) => {
        sessoes.push(params);
        if (falharSessao) throw new Error("QA084: falha simulada ao criar a sessao");
        return { id: `cs_qa084_${sessoes.length}`, url: "https://checkout.stripe.test/qa084" };
      },
    },
  },
  coupons: {
    retrieve: async (id: string) => {
      if (cupomExiste) return { id };
      throw new Error("No such coupon: " + id);
    },
    create: async (params: any) => {
      cuponsStripe.push(params);
      return params;
    },
  },
};

process.env.STRIPE_SECRET_KEY = "sk_test_qa084_falso_nunca_vai_a_rede";

import { NextRequest } from "next/server";
import { prisma } from "@/lib/db";
import { confirmarPorSessao, descartarPorSessao } from "@/lib/coupon-redemption";

const POST_previa = require("@/app/api/patient/coupons/preview/route").POST;
const POST_consulta = require("@/app/api/patient/appointments/[id]/checkout/route").POST;
const POST_pacote = require("@/app/api/patient/packages/checkout/route").POST;
const POST_plano = require("@/app/api/patient/treatment-plans/checkout/route").POST;
const POST_assinatura = require("@/app/api/patient/membership/subscribe/route").POST;

const req = (url: string, body: any) =>
  new NextRequest(url, {
    method: "POST",
    body: JSON.stringify(body),
    headers: { "content-type": "application/json", "x-platform": "mobile" },
  });

const previa = async (body: any) => {
  const r = await POST_previa(req("http://localhost:4100/api/patient/coupons/preview", body));
  return { status: r.status, json: await r.json() };
};
const consulta = async (id: string, body: any) => {
  const r = await POST_consulta(req(`http://localhost:4100/api/patient/appointments/${id}/checkout`, body), {
    params: { id },
  });
  return { status: r.status, json: await r.json() };
};
const ultimaSessao = () => sessoes[sessoes.length - 1];
const ultimaSessaoId = () => `cs_qa084_${sessoes.length}`;
const unitAmount = (s: any) => s?.line_items?.[0]?.price_data?.unit_amount ?? null;

const resgates = (code: string, patientId?: string) =>
  prisma.couponRedemption.findMany({
    where: { coupon: { code }, ...(patientId ? { patientId } : {}) },
    orderBy: { createdAt: "asc" },
  });
const limpar = async (code: string) => {
  await prisma.couponRedemption.deleteMany({ where: { coupon: { code } } });
};
/** Conta so o que pertence a clinica de teste: o Postgres local e compartilhado entre worktrees. */
const contarTudo = () =>
  prisma.couponRedemption.count({ where: { coupon: { clinicId: dados.clinicId } } });
const limparTudo = () =>
  prisma.couponRedemption.deleteMany({ where: { coupon: { clinicId: dados.clinicId } } });

beforeEach(async () => {
  await limparTudo();
  sessoes.length = 0;
  cuponsStripe.length = 0;
  falharSessao = false;
  cupomExiste = false;
  atual = A;
});

const ver = (rotulo: string, v: any) =>
  console.log(`   ${rotulo}: ${typeof v === "string" ? v : JSON.stringify(v)}`);

describe("4.1 — o valor cobrado e o valor mostrado", () => {
  it("consulta: previa 80 e unit_amount 8000", async () => {
    await limpar("QA084-20");
    const p = await previa({ code: "QA084-20", scope: "CONSULTATION" });
    ver("previa", p.json);
    const c = await consulta(dados.appointments.c1, { couponCode: "QA084-20" });
    ver("status do checkout", c.status);
    ver("unit_amount enviado a Stripe", unitAmount(ultimaSessao()));
    ver("metadata", ultimaSessao().metadata);
    const linhas = await resgates("QA084-20", A);
    ver(
      "CouponRedemption",
      linhas.map((l) => ({
        orig: l.originalAmount,
        desc: l.discountAmount,
        final: l.finalAmount,
        sessao: l.stripeSessionId,
        confirmado: l.confirmedAt,
      }))
    );
    expect(p.json.final).toBe(80);
    expect(unitAmount(ultimaSessao())).toBe(8000);
    expect(unitAmount(ultimaSessao())).toBe(Math.round(p.json.final * 100));
    expect(linhas).toHaveLength(1);
    expect(linhas[0].finalAmount).toBe(80);
    expect(linhas[0].stripeSessionId).toBe(ultimaSessaoId());
  });

  it("sem cupom: cobra o preco cheio (100 -> 10000)", async () => {
    await limpar("QA084-20");
    const c = await consulta(dados.appointments.c2, {});
    ver("unit_amount", unitAmount(ultimaSessao()));
    ver("metadata", ultimaSessao().metadata);
    expect(c.status).toBe(200);
    expect(unitAmount(ultimaSessao())).toBe(10000);
    expect(ultimaSessao().metadata.couponCode).toBeUndefined();
    expect(await contarTudo()).toBe(0);
  });

  it("valor fixo de 15: unit_amount 8500", async () => {
    await limpar("QA084-FIX15");
    const p = await previa({ code: "QA084-FIX15", scope: "CONSULTATION" });
    const c = await consulta(dados.appointments.c3, { couponCode: "qa084-fix15" });
    ver("previa", p.json);
    ver("unit_amount", unitAmount(ultimaSessao()));
    expect(c.status).toBe(200);
    expect(unitAmount(ultimaSessao())).toBe(8500);
    expect(unitAmount(ultimaSessao())).toBe(Math.round(p.json.final * 100));
    await limpar("QA084-FIX15");
  });
});

    await limpar("QA084-20");
    const c = await consulta(dados.appointments.c4, {
      couponCode: "QA084-20",

describe("4.5 — idempotencia e expiracao", () => {
  it("confirmarPorSessao duas vezes confirma uma; descartar nao apaga o confirmado", async () => {
    await limpar("QA084-20");
    await consulta(dados.appointments.c8, { couponCode: "QA084-20" });
    const sessao = ultimaSessaoId();
    const primeiro = await confirmarPorSessao(sessao);
    const segundo = await confirmarPorSessao(sessao);
    ver("confirmarPorSessao 1a vez (linhas alteradas)", primeiro);
    ver("confirmarPorSessao 2a vez (linhas alteradas)", segundo);
    const apagadas = await descartarPorSessao(sessao);
    const aindaLa = await resgates("QA084-20", A);
    ver("descartarPorSessao sobre o confirmado (linhas apagadas)", apagadas);
    ver("resgate continua no banco", aindaLa.map((l) => ({ sessao: l.stripeSessionId, confirmado: l.confirmedAt })));
    expect(primeiro).toBe(1);
    expect(segundo).toBe(0);
    expect(apagadas).toBe(0);
    expect(aindaLa).toHaveLength(1);
    expect(aindaLa[0].confirmedAt).not.toBeNull();
    await limpar("QA084-20");
  });

  it("sessao expirada sem pagamento devolve a vaga", async () => {
    await limpar("QA084-ONE");
    await consulta(dados.appointments.c1, { couponCode: "QA084-ONE" });
    const sessao = ultimaSessaoId();
    const antes = await resgates("QA084-ONE");
    atual = B;
    const bloqueado = await previa({ code: "QA084-ONE", scope: "CONSULTATION" });
    atual = A;
    const apagadas = await descartarPorSessao(sessao);
    atual = B;
    const liberado = await previa({ code: "QA084-ONE", scope: "CONSULTATION" });
    atual = A;
    ver("reserva do A", antes.map((l) => ({ sessao: l.stripeSessionId, confirmado: l.confirmedAt })));
    ver("previa do B com a vaga ocupada", bloqueado.json);
    ver("descartarPorSessao (linhas apagadas)", apagadas);
    ver("previa do B depois de a sessao expirar", liberado.json);
    expect(antes).toHaveLength(1);
    expect(bloqueado.json.reason).toBe("limit_reached");
    expect(apagadas).toBe(1);
    expect(liberado.json.ok).toBe(true);
    await limpar("QA084-ONE");
  });
});

describe("4.6 — limite esgotado entre a previa e o pagamento", () => {
  it("o segundo paciente e recusado com frase, nao com 500", async () => {
    await limpar("QA084-ONE");
    const pA = await previa({ code: "QA084-ONE", scope: "CONSULTATION" });
    atual = B;
    const pB = await previa({ code: "QA084-ONE", scope: "CONSULTATION" });
    atual = A;
    const cA = await consulta(dados.appointments.c2, { couponCode: "QA084-ONE" });
    atual = B;
    const cB = await consulta(dados.appointmentB, { couponCode: "QA084-ONE" });
    atual = A;
    ver("previa de A e de B antes de qualquer resgate", [pA.json.final, pB.json.final]);
    ver("A paga", { status: cA.status, unit_amount: unitAmount(ultimaSessao()) });
    ver("B tenta pagar", { status: cB.status, body: cB.json });
    const linhas = await resgates("QA084-ONE");
    ver("resgates do cupom de uso unico", linhas.map((l) => ({ paciente: l.patientId === A ? "A" : "B", final: l.finalAmount })));
    expect(cA.status).toBe(200);
    expect(cB.status).toBe(409);
    expect(cB.json.reason).toBe("limit_reached");
    expect(cB.json.error).toBe("That code has reached its limit.");
    expect(cB.json.errorPt).toBe("Esse codigo atingiu o limite de usos.".replace("codigo", "código"));
    expect(linhas).toHaveLength(1);
    await limpar("QA084-ONE");
  });
});
